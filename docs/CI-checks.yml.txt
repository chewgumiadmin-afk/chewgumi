name: 화면 점검

on:
  push:
    branches: [main]
  pull_request:
  schedule:
    - cron: '0 0 * * *'      # 매일 오전 9시(한국)
  workflow_dispatch:

jobs:
  static:
    name: 정적 점검 (코드만 읽기)
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: '3.12' }

      - name: 디자인·세션·메뉴·동작
        run: python3 tools/page-audit.py

      - name: 끊긴 함수·링크·서버함수·DB표
        run: python3 tools/flow-trace.py

      - name: 눌러도 아무 일 없는 단추
        run: python3 tools/dead-action.py

      - name: 이미지 짝맞춤
        run: python3 tools/img-check.py

  live:
    name: 브라우저 점검 (실제로 눌러보기)
    runs-on: ubuntu-latest
    needs: static
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: '3.12' }

      - name: 브라우저 준비
        run: |
          pip install playwright
          python3 -m playwright install --with-deps chromium

      - name: 배포 기다리기
        run: sleep 90

      - name: 손님 화면
        run: python3 tools/live-check.py
